function checkUrlParams(){
  const p=new URLSearchParams(location.search);
  const hash=location.hash.replace('#','');
  if(p.get('product')){openProduct(parseInt(p.get('product')),p.get('ar')==='1');return;}
  if(p.get('cat')){showView('shop');renderShop(p.get('cat'));return;}
  if(p.get('blog')){openBlogPost(parseInt(p.get('blog')));return;}
  const views=['about','faq','terms','shipping','returns','customisation','login','account','blog'];
  if(views.includes(hash)){
    if(hash==='blog'){showView('blog-list');renderBlogList();return;}
    showView(hash);
    if(hash==='account')initAccount();
    if(hash==='faq')renderFaq();
    if(hash==='login')switchAuthTab('login');
    return;
  }
  showView('shop');
  renderShop('all');
  if(hash==='cart')openCart();
}

function getCart(){return JSON.parse(localStorage.getItem('kwCart')||'[]');}

  const n=getCart().reduce((s,i)=>s+i.qty,0);

function addToCart(id){
  const p=getProducts().find(x=>x.id===id);
  if(!p)return;
  const cart=getCart();
  const ex=cart.find(i=>i.id===id&&!i.customColor&&!i.customSize);

  saveCart(cart);updateCartBadge();
  toast('Added — '+p.name);
}

function getUser(){const u=localStorage.getItem('kwUser');return u?JSON.parse(u):null;}

  ['login','signup','forgot'].forEach(t=>{
    const f=document.getElementById(t+'-form');
    const b=document.getElementById('tab-'+t);

    if(b)b.classList.toggle('active',t===tab);
  });
}

function doLogin(){
  const email=document.getElementById('l-email').value.trim();
  const pass=document.getElementById('l-pass').value;
  const err=document.getElementById('l-err');
  if(!email||!pass){err.textContent='Fill all fields';err.style.display='block';return;}
  const d=db();
  const users=d.users||[];
  const user=users.find(u=>u.email===email&&u.password===pass);
  if(!user){err.textContent='Incorrect email or password';err.style.display='block';return;}
  localStorage.setItem('kwUser',JSON.stringify(user));
  updateNavAuth();showView('account');initAccount();
  toast('Welcome back, '+user.name+'!');
}

function doSignup(){
  const name=document.getElementById('s-name').value.trim();
  const email=document.getElementById('s-email').value.trim();
  const phone=document.getElementById('s-phone').value.trim();
  const pass=document.getElementById('s-pass').value;
  const err=document.getElementById('s-err');
  if(!name||!email||!pass){err.textContent='Fill all required fields';err.style.display='block';return;}
  if(pass.length<6){err.textContent='Password must be at least 6 characters';err.style.display='block';return;}
  const d=db();
  if(!d.users)d.users=[];
  if(d.users.find(u=>u.email===email)){err.textContent='Email already registered';err.style.display='block';return;}
  const user={id:Date.now(),name,email,phone,password:pass,joined:new Date().toLocaleDateString('en-IN')};
  d.users.push(user);sdb(d);
  localStorage.setItem('kwUser',JSON.stringify(user));
  updateNavAuth();showView('account');initAccount();
  toast('Welcome to Kanchwale, '+name+'!');
}
