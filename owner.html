      <td>${u.joined||'-'}</td>
      <td>${(d.orders||[]).filter(o=>o.userId===u.email).length}</td>
    </tr>`).join(''):'<tr><td colspan="5" style="text-align:center;color:rgba(245,240,234,.26);padding:24px;">No registered users yet</td></tr>');
  // Sessions
  const sl=document.getElementById('sessions-list');
  const sessions=(d.visitors.sessions||[]).slice(0,10);
  sl.innerHTML=sessions.length?
    '<div class="vis-row hdr"><div>Date & Time</div><div>Page</div><div>Browser Info</div></div>'+
    sessions.map(s=>`<div class="vis-row"><div>${s.date}</div><div>${s.page||'Home'}</div><div style="font-size:10px;">${(s.ua||'').substring(0,50)}</div></div>`).join('')
    :'<p style="font-size:13px;color:rgba(245,240,234,.26);">No sessions recorded yet.</p>';
}

// ── FAQ ──────────────────────────────────────────────────
function saveFaq(){
  const q=document.getElementById('faq-q').value.trim();
  const a=document.getElementById('faq-a').value.trim();
  if(!q||!a){toast('Question and answer required',true);return;}
  const d=ensureDB();
  if(!d.faq)d.faq=[];
  d.faq.push({id:Date.now(),q,a});
  sdb(d);
  document.getElementById('faq-q').value='';
  document.getElementById('faq-a').value='';
  renderFaqs();toast('FAQ added!');
}

function renderFaqs(){
  const d=ensureDB();
  const t=document.getElementById('faq-table');
  const count=document.getElementById('faq-count');
  if(count)count.textContent=(d.faq||[]).length;
  t.innerHTML=`<tr><th>Question</th><th>Actions</th></tr>`+
    (d.faq&&d.faq.length?d.faq.map(f=>`<tr>
      <td style="max-width:400px;">${f.q}</td>
      <td><button class="btn-del" onclick="deleteFaq(${f.id})">Delete</button></td>
    </tr>`).join(''):'<tr><td colspan="2" style="text-align:center;color:rgba(245,240,234,.26);padding:24px;">No FAQs yet</td></tr>');
}

function deleteFaq(id){
  const d=ensureDB();d.faq=d.faq.filter(f=>f.id!==id);sdb(d);renderFaqs();toast('FAQ deleted');
}

function changePassword(){
  const cur=document.getElementById('cur-pass').value;
  const nw=document.getElementById('new-pass').value;
  const cn=document.getElementById('con-pass').value;
  const d=ensureDB();
  if(cur!==(d.adminPassword||'kanchwale@2025')){toast('Current password incorrect',true);return;}
  if(nw.length<6){toast('New password must be at least 6 characters',true);return;}
  if(nw!==cn){toast('Passwords do not match',true);return;}
  d.adminPassword=nw;sdb(d);
  document.getElementById('cur-pass').value='';
  document.getElementById('new-pass').value='';
  document.getElementById('con-pass').value='';
  toast('Password updated successfully!');
}
